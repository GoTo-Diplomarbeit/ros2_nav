# Python dependencies that come from pip (setup.sh installs these into ~/venvs/go2).
#
# NOT listed here on purpose: rclpy, geometry_msgs, nav_msgs, sensor_msgs, ...
# They ship with ROS 2 Jazzy (apt: ros-jazzy-desktop) and are only importable after
# sourcing /opt/ros/jazzy/setup.bash. Install the venv through setup.sh so it carries
# that environment.
#
# PITFALL: ~/venvs/go2 is only what PyCharm uses. Nodes started with "ros2 launch" run
# with the SYSTEM python (/usr/bin/python3) and never see the venv. setup.sh therefore
# also installs the packages the nodes import at runtime for the system python
# (fastapi, uvicorn, starlette, zeroconf via apt; ultralytics via pip --user).

# ROS 2 Jazzy's compiled packages are built against NumPy 1.x
numpy<2
# colcon-core needs setuptools < 80
setuptools<80
opencv-python<4.12

# Real-robot backend (Go2Backend). Imported lazily, so the simulator does not need it.
# Building cyclonedds needs the CycloneDDS that ROS 2 ships: source ROS before installing.
unitree_sdk2py @ git+https://github.com/unitreerobotics/unitree_sdk2_python.git

# Tests (src/go2_nav/test)
pytest

# App communication (go2_nav/communication/camera_node.py): web server for the app
# (fastapi + starlette, served by uvicorn) and network discovery (zeroconf).
fastapi
zeroconf
uvicorn[standard]
aiortc
aiohttp

# Person and face recognition (go2_nav/vision/image_recognition.py): YOLO models.
# Large: pulls in PyTorch (a few GB).
ultralytics

# Needed for communication across networks
iroh
